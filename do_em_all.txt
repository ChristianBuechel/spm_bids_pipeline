function do_em_all
% Example wrapper: runs the steps of the pipeline for the subjects in all_sub_ids.
% Copy it to your "scripts" folder, rename it to do_em_all.m and set the flags of the steps you want to run to 1.
% What the steps do and which images they work on is set in get_study_specs
% (vars.preproc.steps, vars.average_T1, ...), see README.md.

addpath(genpath('c:\your_path\spm_bids_pipeline\')); % specify path to bids pipeline

% define subjects that should be included
all_sub_ids     = [1:5];

% data
do_import       = 0;     % download and convert the DICOMs (dry_run = 1: only print the plan and fetch the first files)
dry_run         = 0;
do_averageT1    = 0;     % average the T1w of a session (vars.nT1 >= 2) or of all sessions (vars.average_T1.across_sessions = 1), BEFORE the alignment
do_align        = 0;     % coarse alignment of all images of each session to template space
do_shift        = 0;     % only if images are far from template space and the alignment fails (4x4 matrix)

% preprocessing: optional NORDIC first, then one of the two workflows
do_nordic       = 0;     % denoise magnitude (+ phase as input), writes n<epi>.nii; needs 'nordic' in vars.preproc.steps
do_preproc_nlin = 0;     % slice timing, realignment, nonlinear coregistration EPI --> T1 (standard)
do_preproc_vdm  = 0;     % the same with a fieldmap (vdm) and linear coregistration
do_smooth       = 0;     % only with 'smooth' in vars.preproc.steps: smooths the resliced images (s<..>)

% group and analysis
do_means        = 0;     % group means of the skull-stripped T1, GM and mean EPI in template space (at least 2 subjects)
do_analysis     = 0;
w_ana           = [1];   % which analyses of the analysis struct array in get_study_specs


if do_import == 1
    sbp_import_data(all_sub_ids,dry_run);
end

if do_averageT1 == 1
    sbp_average_T1(all_sub_ids);
end

if do_align == 1
    sbp_align_images(all_sub_ids);
end

if do_shift == 1
    sbp_shift_images(all_sub_ids);
end

if do_nordic == 1
    sbp_nordic(all_sub_ids);
end


if do_preproc_nlin == 1
    % slice timing correction
    sbp_slice_timing(all_sub_ids);

    % realign part 1 of 2
    sbp_realign_1_2(all_sub_ids);

    % nonlinear coregistration to get from EPI space to T1 space (also segments the T1)
    % (for VASO slab data: sbp_coreg_nlin_vaso_T1 instead)
    sbp_coreg_nlin(all_sub_ids);

    % skull stripping
    sbp_skullstrip(all_sub_ids);

    % spatial normalization of the T1 to template space using DARTEL
    sbp_create_dartel(all_sub_ids);

    % create the warp fields EPI --> T1 --> template space
    sbp_create_trans(all_sub_ids);

    % create the brain mask for the 1st level GLM (and for the weighting of realign 2)
    sbp_create_mask(all_sub_ids);

    % realign part 2 of 2 (with the mask, reslices all images)
    sbp_realign_2_2(all_sub_ids);

    % warp the skull-stripped T1 and the mean EPI to template space
    sbp_warp(all_sub_ids);

    % 6 WM and 6 CSF noise regressors (and 6 each from the posterior part of the lateral ventricles)
    sbp_create_noise(all_sub_ids);
end


if do_preproc_vdm == 1
    sbp_slice_timing(all_sub_ids);
    sbp_realign_1_2(all_sub_ids);

    % voxel displacement map from the fieldmap, linear coregistration EPI --> T1, T1 segmentation
    sbp_create_vdm_fmap(all_sub_ids);
    sbp_coreg_lin(all_sub_ids);
    sbp_seg_T1(all_sub_ids);

    sbp_skullstrip(all_sub_ids);
    sbp_create_dartel(all_sub_ids);
    sbp_create_trans_vdm(all_sub_ids);
    sbp_create_mask(all_sub_ids);
    sbp_warp_vdm(all_sub_ids);

    % realign part 2 of 2 and apply the vdm (reslices all images)
    sbp_realign_2_2_vdm(all_sub_ids);
    sbp_apply_vdm(all_sub_ids);

    sbp_create_noise(all_sub_ids);
end


if do_smooth == 1
    sbp_smooth(all_sub_ids);
end

if do_means == 1
    sbp_create_means(all_sub_ids);
end

if do_analysis == 1
    for ii = 1:numel(w_ana)
        sbp_analyses(all_sub_ids,w_ana(ii));
    end
end
end
